   WHEEL OF FORTUNE
   ============================================ */
function buildWheelSVG() {
  const svg = document.getElementById('wheel-svg');
  if (!svg) return;
  svg.innerHTML = '';
  const cx = 100, cy = 100, r = 95;
  const n = WHEEL_PRIZES.length;
  const step = (Math.PI * 2) / n;

  const outer = document.createElementNS('http://www.w3.org/2000/svg','circle');
  outer.setAttribute('cx', cx); outer.setAttribute('cy', cy);
  outer.setAttribute('r', r + 3);
  outer.setAttribute('fill', '#fff');
  outer.setAttribute('stroke', '#a78bfa');
  outer.setAttribute('stroke-width', '3');
  svg.appendChild(outer);

  for (let i = 0; i < n; i++) {
    const startA = -Math.PI/2 + i * step;
    const endA = startA + step;
    const x1 = cx + r * Math.cos(startA);
    const y1 = cy + r * Math.sin(startA);
    const x2 = cx + r * Math.cos(endA);
    const y2 = cy + r * Math.sin(endA);
    const path = document.createElementNS('http://www.w3.org/2000/svg','path');
    const largeArc = step > Math.PI ? 1 : 0;
    const d = `M ${cx} ${cy} L ${x1} ${y1} A ${r} ${r} 0 ${largeArc} 1 ${x2} ${y2} Z`;
    path.setAttribute('d', d);
    path.setAttribute('fill', WHEEL_PRIZES[i].color);
    path.setAttribute('stroke', '#fff');
    path.setAttribute('stroke-width', '2');
    svg.appendChild(path);

    const midA = startA + step / 2;
    const tr = r * 0.62;
    const tx = cx + tr * Math.cos(midA);
    const ty = cy + tr * Math.sin(midA);
    const text = document.createElementNS('http://www.w3.org/2000/svg','text');
    text.setAttribute('x', tx);
    text.setAttribute('y', ty);
    text.setAttribute('text-anchor', 'middle');
    text.setAttribute('dominant-baseline', 'middle');
    text.setAttribute('font-size', WHEEL_PRIZES[i].label.length > 6 ? '9' : '12');
    const degrees = (midA * 180 / Math.PI) + 90;
    text.setAttribute('transform', `rotate(${degrees}, ${tx}, ${ty})`);
    const parts = WHEEL_PRIZES[i].label.split(' ');
    if (parts.length > 1) {
      const t1 = document.createElementNS('http://www.w3.org/2000/svg','tspan');
      t1.setAttribute('x', tx); t1.setAttribute('dy', '-0.5em'); t1.textContent = parts[0];
      const t2 = document.createElementNS('http://www.w3.org/2000/svg','tspan');
      t2.setAttribute('x', tx); t2.setAttribute('dy', '1.1em'); t2.textContent = parts.slice(1).join(' ');
      text.appendChild(t1); text.appendChild(t2);
    } else {
      text.textContent = WHEEL_PRIZES[i].label;
    }
    svg.appendChild(text);
  }
}

function canSpinWheel() { return (Date.now() - lastWheelSpin) >= WHEEL_COOLDOWN_MS; }
function canClaimDaily() { return (Date.now() - lastDailyClaim) >= DAILY_COOLDOWN_MS; }
function formatCountdown(ms) {
  if (ms <= 0) return 'Ready!';
  const totalSec = Math.floor(ms / 1000);
  const h = Math.floor(totalSec / 3600);
  const m = Math.floor((totalSec % 3600) / 60);
  const s = totalSec % 60;
  return `${String(h).padStart(2,'0')}:${String(m).padStart(2,'0')}:${String(s).padStart(2,'0')}`;
}

function spinWheel() {
  if (wheelSpinning) return;
  if (!canSpinWheel()) { Sound.playInvalid(); return; }
  wheelSpinning = true;
  const spinBtn = document.getElementById('wheel-spin-btn');
  spinBtn.disabled = true;
  spinBtn.textContent = '...';
  Sound.playSpin();

  const n = WHEEL_PRIZES.length;
  const step = 360 / n;
  const prizeIdx = rand(0, n - 1);
  const targetAngle = -(prizeIdx * step + step / 2);
  const currentMod = ((wheelCurrentRotation % 360) + 360) % 360;
  const targetMod = ((targetAngle % 360) + 360) % 360;
  let deltaForward = (targetMod - currentMod + 360) % 360;
  const extraSpins = 5;
  const totalDelta = deltaForward + 360 * extraSpins;
  wheelCurrentRotation += totalDelta;

  const svg = document.getElementById('wheel-svg');
  svg.style.transform = `rotate(${wheelCurrentRotation}deg)`;

  setTimeout(() => {
    const prize = WHEEL_PRIZES[prizeIdx];
    applyWheelPrize(prize);
    lastWheelSpin = Date.now();
    saveData();
    Sound.playWin();
    wheelSpinning = false;
    spinBtn.disabled = true;
    spinBtn.classList.add('disabled');
    spinBtn.textContent = 'WAIT';
    updateEventUI();
  }, 4100);
}

function applyWheelPrize(prize) {
  if (prize.type === 'coins') {
    coins += prize.amount;
    updateCoins();
    showRewardToast(` +${prize.amount} `);
  } else if (prize.type === 'infinity') {
    infinityTimeUntil = Date.now() + prize.amount * 1000;
    updateTimer();
    showRewardToast(` INFINITY TIME 1H!`);
  }
}

   2.2 DAILY CHALLENGE MODAL (NEW)
   ============================================ */
function showChallengeModal() {
  const today = getUTCDateKey();
  // Reset today's tracking if the calendar day rolled over.
  if (challengeDate !== today) { challengeDate = today; challengeBestScore = 0; challengeAttempts = 0; saveData(); }

  document.getElementById('challenge-date').textContent = `Today's board · ${today} (UTC)`;
  document.getElementById('challenge-best-score').textContent = challengeBestScore;
  document.getElementById('challenge-attempts').textContent = challengeAttempts;
  document.getElementById('challenge-countdown').textContent = ' Resets in ' + formatCountdown(msUntilUTCMidnight());

  /* Ghost toggle only meaningful if a best replay exists for today. */
  const ghostToggle = document.getElementById('ghost-toggle');
  const hasTodayGhost = bestChallengeReplay && bestChallengeReplay.date === today && bestChallengeReplay.log.length;
  ghostToggle.disabled = !hasTodayGhost;
  ghostToggle.checked = hasTodayGhost ? ghostEnabled : false;
  ghostToggle.parentElement.style.opacity = hasTodayGhost ? '1' : '0.5';

  /* Watch-replay button for today's best challenge run. */
  const rBtn = document.getElementById('challenge-replay-btn');
  rBtn.style.display = hasTodayGhost ? 'block' : 'none';

  document.getElementById('challenge-modal').classList.remove('hidden');
  document.getElementById('app').classList.add('blurred');
  if (gameActive) stopTimer();
}
function hideChallengeModal() {
  document.getElementById('challenge-modal').classList.add('hidden');
  document.getElementById('app').classList.remove('blurred');
  if (gameActive) startTimer();
}

/* ============================================
   EVENT CAROUSEL SCROLL DOTS
   ============================================ */
function setupEventCarousel() {
  const carousel = document.getElementById('events-carousel');
  const dots = document.querySelectorAll('.event-dot');
  if (!carousel) return;
  carousel.addEventListener('scroll', () => {
    const idx = Math.round(carousel.scrollLeft / carousel.clientWidth);
    dots.forEach((d, i) => d.classList.toggle('active', i === idx));
  });
  dots.forEach(dot => {
    dot.addEventListener('click', () => {
      const idx = parseInt(dot.dataset.idx);
      carousel.scrollTo({ left: idx * carousel.clientWidth, behavior: 'smooth' });
    });
  });
}

/* ============================================
   Game Control
   ============================================ */
/* Unified run starter that both freeplay and challenge funnel through.
   `mode` = 'freeplay' | 'challenge'. Challenge arms the seeded RNG so the
   generated board & capsules match everyone else's for the day (Section 2.2). */
function startGame(mode = 'freeplay') {
  Sound.init(); Sound.resume(); Sound.playStart();
  hideStartScreen();
  hideChallengeModal();
  currentMode = mode;

  /* 2.2 Arm/disarm the seeded RNG for the initial board generation. */
  if (mode === 'challenge') {
    const key = getUTCDateKey();
    currentRunSeed = hashStringToSeed('EmberChain-' + key);
    seededRNG = mulberry32(currentRunSeed);
    document.getElementById('board-frame').classList.add('challenge-mode');
  } else {
    currentRunSeed = null;
    seededRNG = null;
    document.getElementById('board-frame').classList.remove('challenge-mode');
  }

  score = 0;
  /* 2.2 Challenge start time is fixed & identical for everyone (fair). Pets
     that grant bonus start time are disabled in challenge to keep it even. */
  let initialTime = GAME_TIME;
  if (mode === 'freeplay') {
    if (currentPetId === 'unicorn') initialTime += 10;
    if (currentPetId === 'wizard_dog') initialTime += 30;
  }
  timeLeft = initialTime;
  combo = 1; processing = false; gameActive = true; isReplaying = false;
  resetFlow();
  bestNearMiss = 0;
  lastAchievedMatch = 0;
  capsulesFilledThisRun = 0;

  /* 2.3 Fresh replay recording for this run. */
  replayLog = [];
  runStartTime = Date.now();

  updateScore(); updateTimer(); updateCombo(); updateCoins(); updateXPDisplay(); updateBest();
  calculateLayout();
  board = generateBoard();

  /* 2.3 For freeplay replays we need the exact starting board (challenge
     replays regenerate from the seed instead). Snapshot before play. */
  currentInitialBoard = board.map(row => row.slice());

  /* 2.2 Disarm the seed now — cascades/refills use Math.random so live play
     and replay stay consistent (both fill non-deterministically after the
     seeded initial board). The initial layout is what's shared. */
  seededRNG = null;

  renderFullBoard();
  initCapsules();

  /* 2.3 Arm the ghost overlay if the player opted in and has a ghost. */
  const today = getUTCDateKey();
  if (mode === 'challenge' && ghostEnabled && bestChallengeReplay &&
      bestChallengeReplay.date === today && bestChallengeReplay.log.length) {
    ghostLog = bestChallengeReplay.log;
    startGhostOverlay();
  } else {
    ghostLog = null;
  }

  startTimer();
  startFlowDecay();
  startHintTimer();
  createConfetti();
}
function restartGame() {
  hideGameMenu(); hideTimeUpModal();
  clearTimeout(hintTimeout); stopTimer(); stopFlowDecay(); stopGhostOverlay();
  boardEl.innerHTML = ''; effectsLayer.innerHTML = '';
  document.getElementById('board-inner').classList.remove('replay-mode');
  startGame(currentMode);
}
function endGame() {
  gameActive = false; isReplaying = false;
  stopTimer(); stopFlowDecay(); stopGhostOverlay(); clearTimeout(hintTimeout);
  hideGameMenu(); hideTimeUpModal();
  document.getElementById('board-inner').classList.remove('replay-mode');
  document.getElementById('board-frame').classList.remove('challenge-mode');
  showStartScreen();
}
function buyTime() {
  /* Guardrail: never allow buying time in the Daily Challenge (Section 7). */
  if (currentMode === 'challenge') return;
  if (coins >= 20) {
    coins -= 20; timeLeft += 15;
    updateCoins(); updateTimer();
    Sound.playClick(); Sound.playStart();
    hideTimeUpModal();
    gameActive = true; processing = false;
    startTimer(); startFlowDecay(); startHintTimer();
    saveData();
  } else {
    const btn = document.getElementById('buy-time-btn');
    btn.style.animation = 'shake 0.35s ease';
    setTimeout(() => btn.style.animation = '', 400);
  }
}
function createConfetti() {
  document.querySelectorAll('.confetti').forEach(el => el.remove());
  const app = document.getElementById('app');
  const colors = ['#ff6b9d', '#ffd93d', '#4ade80', '#60a5fa', '#f9a8d4', '#ffb347'];
  for (let i = 0; i < 15; i++) {
    const c = document.createElement('div');
    c.className = 'confetti';
    c.style.left = rand(5, 95) + '%';
    c.style.background = pick(colors);
    c.style.width = rand(3, 7) + 'px'; c.style.height = rand(3, 7) + 'px';
    c.style.animationDuration = rand(8, 18) + 's';
    c.style.animationDelay = rand(0, 15) + 's';
    if (Math.random() > 0.5) c.style.borderRadius = '2px';
    app.appendChild(c);
  }
}
