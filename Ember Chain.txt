function mulberry32(seed) {
  let a = seed >>> 0;
  return function() {
    a |= 0; a = (a + 0x6D2B79F5) | 0;
    let t = Math.imul(a ^ (a >>> 15), 1 | a);
    t = (t + Math.imul(t ^ (t >>> 7), 61 | t)) ^ t;
    return ((t ^ (t >>> 14)) >>> 0) / 4294967296;
  };
}
function hashStringToSeed(str) {
  let h = 2166136261 >>> 0;
  for (let i = 0; i < str.length; i++) {
    h ^= str.charCodeAt(i);
    h = Math.imul(h, 16777619);
  }
  return h >>> 0;
}

    const t = this.ctx.currentTime + delay;
    const osc = this.ctx.createOscillator();
    const gain = this.ctx.createGain();
    osc.type = type; osc.frequency.setValueAtTime(freq, t);
    gain.gain.setValueAtTime(vol, t);
    gain.gain.exponentialRampToValueAtTime(0.001, t + dur);
    osc.connect(gain); gain.connect(this.masterGain);
    osc.start(t); osc.stop(t + dur);
  },

    if (!this.ctx || !this.enabled) return;
    const t = this.ctx.currentTime;
    const osc = this.ctx.createOscillator();
    const gain = this.ctx.createGain();
    osc.type = type;
    osc.frequency.setValueAtTime(f1, t);
    osc.frequency.exponentialRampToValueAtTime(f2, t + dur);
    gain.gain.setValueAtTime(vol, t);
    gain.gain.exponentialRampToValueAtTime(0.001, t + dur);
    osc.connect(gain); gain.connect(this.masterGain);
    osc.start(t); osc.stop(t + dur);
  },

    if (data.petsOwned) {
      data.petsOwned.forEach(id => {
        const pet = petsList.find(p => p.id === id);
        if (pet) pet.owned = true;
      });
    }
    if (data.currentPetId) currentPetId = data.currentPetId;
    if (data.gearOwned) {
      data.gearOwned.forEach(id => {
        const gear = gearList.find(g => g.id === id);
        if (gear) gear.owned = true;
      });
    }
    if (Array.isArray(data.gearLoadout)) {
      for (let i = 0; i < GEAR_SLOTS; i++) gearLoadout[i] = data.gearLoadout[i] || 'none';

function getEquippedGear() {
  const seen = new Set();
  const out = [];
  gearLoadout.forEach(id => {
    if (id && id !== 'none' && !seen.has(id)) {
      seen.add(id);
      const g = gearList.find(x => x.id === id);
      if (g) out.push(g);
    }
  });
  return out;
}

function computeLoadoutEffects() {
  const agg = {
    scoreMult: 1, coinMult: 1, xpMult: 1, comboMult: 1,
    bombChance: 0, capsuleTimeAdd: 0, timePerMatchAdd: 0, timerSlow: false,
  };
  getEquippedGear().forEach(g => {
    const e = g.effects || {};
    if (e.scoreMultAdd) agg.scoreMult += e.scoreMultAdd;
    if (e.coinMultAdd) agg.coinMult += e.coinMultAdd;
    if (e.xpMultAdd) agg.xpMult += e.xpMultAdd;
    if (e.comboMultAdd) agg.comboMult += e.comboMultAdd;
    if (e.bombChanceAdd) agg.bombChance += e.bombChanceAdd;
    if (e.capsuleTimeAdd) agg.capsuleTimeAdd += e.capsuleTimeAdd;
    if (e.timePerMatchAdd) agg.timePerMatchAdd += e.timePerMatchAdd;
    if (e.timerSlow) agg.timerSlow = true;
  });
  agg.scoreMult = Math.max(0.25, agg.scoreMult);
  agg.coinMult  = Math.max(0.25, agg.coinMult);
  agg.xpMult    = Math.max(0.25, agg.xpMult);
  agg.comboMult = Math.max(1,    agg.comboMult);
  return agg;
}

/* Daily login streak */
function checkDailyLogin() {
  const today = new Date().toISOString().split('T')[0];
  if (lastLoginDate !== today) {
    const yesterday = new Date(Date.now() - 86400000).toISOString().split('T')[0];
    if (lastLoginDate === yesterday) loginStreak++;
    else loginStreak = 1;
    lastLoginDate = today;
    const streakReward = Math.min(loginStreak * 10, 100);
    coins += streakReward;
    saveData();
  }
  const streakEl = document.getElementById('streak-display');
  const streakCountEl = document.getElementById('streak-count');
  if (streakEl && loginStreak > 0) {
    streakEl.style.display = 'flex';
    streakCountEl.textContent = loginStreak;
  }
}

function tierIndex(id) { return TIERS.findIndex(t => t.id === id); }
function computeRating() {
  const comboBonus = maxCombo * 120;
  const scoreComponent = Math.floor(bestScore * 0.9);
  return scoreComponent + comboBonus;
}
function tierForRating(rating) {
  let chosen = TIERS[0];
  for (const t of TIERS) if (rating >= t.min) chosen = t;
  return chosen;
}
function ensureCompetitiveWeek() {
  const now = Date.now();
  if (!weekStart) { weekStart = now; return; }
  while (now - weekStart >= WEEK_MS) {
    const snapTier = tierForRating(computeRatingFromScore(weekBestScore));
    weekHistory.unshift({
      week: getUTCDateKey(new Date(weekStart)),
      bestScore: weekBestScore,
      tier: snapTier.id,
    });
    if (weekHistory.length > MAX_WEEK_HISTORY) weekHistory.length = MAX_WEEK_HISTORY;
    weekStart += WEEK_MS;
    weekBestScore = 0;
  }
}
function computeRatingFromScore(s) { return Math.floor(s * 0.9) + maxCombo * 120; }

/* THE single score-submission choke-point (Section 2.1).
   BACKEND NOTE: replace body with an API POST and hydrate from response. */
function submitScore(finalScore, mode) {
  ensureCompetitiveWeek();
  if (finalScore > weekBestScore) weekBestScore = finalScore;
  if (mode === 'freeplay' && finalScore > bestScore) bestScore = finalScore;
  const prevTierIdx = tierIndex(currentTierId);
  playerRating = computeRating();
  const newTier = tierForRating(playerRating);
  currentTierId = newTier.id;
  updateTierBadge();

  if (newIdx > prevTierIdx) {
    Sound.playTierUp();
    showAchievementToast(`${newTier.name} Tier reached!`);
  }
  return { rating: playerRating, tier: newTier };
}

   Here the code just round-trips through localStorage as a demo.

let selectedGiftType = 'spin';
function encodeGift(giftType) {
  // Compact, human-shareable code: EC-<TYPE>-<RANDOM>
  const rnd = Math.random().toString(36).slice(2, 8).toUpperCase();
  const typeCode = giftType === 'coins' ? 'C200' : 'SPIN';
  return `EC-${typeCode}-${rnd}`;
}
function decodeGift(code) {
  const parts = (code || '').trim().toUpperCase().split('-');
  if (parts.length !== 3 || parts[0] !== 'EC') return null;
  if (parts[1] === 'C200') return { type: 'coins', amount: 200 };
  if (parts[1] === 'SPIN') return { type: 'spin' };
  return null;
}

/* Board logic */
function randomType() { return seededInt(NUM_TYPES); }
function generateBoard() {
  let b, attempts = 0;
  do {
    b = [];
    for (let r = 0; r < ROWS; r++) {
      b[r] = [];
      for (let c = 0; c < COLS; c++) {
        let type, tries = 0;
        do {
          type = randomType(); tries++;
        } while (tries < 50 && (
          (c >= 2 && b[r][c-1] === type && b[r][c-2] === type) ||
          (r >= 2 && b[r-1][c] === type && b[r-2][c] === type)
        ));
        b[r][c] = type;
      }
    }
    attempts++;
  } while (!hasValidMoves(b) && attempts < 100);
  return b;
}
function findMatches(b) {
  const matched = new Set(); const groups = [];
  for (let r = 0; r < ROWS; r++) {
    let c = 0;
    while (c < COLS) {
      const type = b[r][c];
      if (type === -1) { c++; continue; }
      let end = c + 1;
      while (end < COLS && b[r][end] === type) end++;
      if (end - c >= 3) {
        const cells = [];
        for (let i = c; i < end; i++) { matched.add(`${r},${i}`); cells.push({ r, c: i }); }
        groups.push({ type, cells });
      }
      c = end;
    }
  }
  for (let c = 0; c < COLS; c++) {
    let r = 0;
    while (r < ROWS) {
      const type = b[r][c];
      if (type === -1) { r++; continue; }
      let end = r + 1;
      while (end < ROWS && b[end][c] === type) end++;
      if (end - r >= 3) {
        const cells = [];
        for (let i = r; i < end; i++) { matched.add(`${i},${c}`); cells.push({ r: i, c }); }
        groups.push({ type, cells });
      }
      r = end;
    }
  }
  return { matched, groups };
}
function hasValidMoves(b) {
  for (let r = 0; r < ROWS; r++) for (let c = 0; c < COLS; c++) {
    if (c < COLS - 1) {
      [b[r][c], b[r][c+1]] = [b[r][c+1], b[r][c]];
      const m = findMatches(b).matched.size > 0;
      [b[r][c], b[r][c+1]] = [b[r][c+1], b[r][c]];
      if (m) return true;
    }
    if (r < ROWS - 1) {
      [b[r][c], b[r+1][c]] = [b[r+1][c], b[r][c]];
      const m = findMatches(b).matched.size > 0;
      [b[r][c], b[r+1][c]] = [b[r+1][c], b[r][c]];
      if (m) return true;
    }
  }
  return false;
}
function findValidMove(b) {
  for (let r = 0; r < ROWS; r++) for (let c = 0; c < COLS; c++) {
    if (c < COLS - 1) {
      [b[r][c], b[r][c+1]] = [b[r][c+1], b[r][c]];
      const m = findMatches(b).matched.size > 0;
      [b[r][c], b[r][c+1]] = [b[r][c+1], b[r][c]];
      if (m) return { r1: r, c1: c, r2: r, c2: c + 1 };
    }
    if (r < ROWS - 1) {
      [b[r][c], b[r+1][c]] = [b[r+1][c], b[r][c]];
      const m = findMatches(b).matched.size > 0;
      [b[r][c], b[r+1][c]] = [b[r+1][c], b[r][c]];
      if (m) return { r1: r, c1: c, r2: r + 1, c2: c };
    }
  }
  return null;
}

/* NEAR-MISS SCAN (Section 3.3) */
function largestPotentialMatch(b) {
  let best = 0;
  const probe = () => {
    const { groups } = findMatches(b);
    let localMax = 0;
    groups.forEach(g => { if (g.cells.length > localMax) localMax = g.cells.length; });
    if (localMax > best) best = localMax;
  };
  for (let r = 0; r < ROWS; r++) for (let c = 0; c < COLS; c++) {
    if (c < COLS - 1) {
      [b[r][c], b[r][c+1]] = [b[r][c+1], b[r][c]];
      probe();
      [b[r][c], b[r][c+1]] = [b[r][c+1], b[r][c]];
    }
    if (r < ROWS - 1) {
      [b[r][c], b[r+1][c]] = [b[r+1][c], b[r][c]];
      probe();
      [b[r][c], b[r+1][c]] = [b[r+1][c], b[r][c]];
    }
  }
  return best;
}

function applyGravity() {
  const fell = [];
  for (let c = 0; c < COLS; c++) {
    let writeRow = ROWS - 1;
    for (let r = ROWS - 1; r >= 0; r--) {
      if (board[r][c] !== -1) {
        if (r !== writeRow) {
          board[writeRow][c] = board[r][c]; board[r][c] = -1;
          candyEls[writeRow][c] = candyEls[r][c]; candyEls[r][c] = null;
          updateCandyPos(candyEls[writeRow][c], writeRow, c);
          fell.push({ r: writeRow, c, dist: writeRow - r });
        }
        writeRow--;
      }
    }
  }
  return fell;
}
function fillEmpty() {
  const spawned = [];
  for (let c = 0; c < COLS; c++) {
    let emptySlots = [];
    for (let r = 0; r < ROWS; r++) if (board[r][c] === -1) emptySlots.push(r);
    for (let i = 0; i < emptySlots.length; i++) {
      const r = emptySlots[i];
      const type = randomType();
      board[r][c] = type;
      const spawnRow = -(emptySlots.length - i);
      const el = createCandyEl(type, spawnRow, c);
      candyEls[r][c] = el;
      el.offsetHeight;
      spawned.push({ el, r, c, delay: i * 30 });
    }
  }
  spawned.forEach(({ el, r, c, delay: d }) => {
    setTimeout(() => updateCandyPos(el, r, c), 10 + d);
  });
  return spawned;
}

/* ============================================
   FLOW STATE logic (Section 3.2)
   --------------------------------------------
   FIX: decay only begins after FLOW_GRACE_MS (5s) of idle, giving the
   player a full 5 seconds before Flow starts easing back to 0.
   ============================================ */
function recomputeFlowTier() {
  let tier = 0, mult = 1;
  for (let i = 0; i < FLOW_TIERS.length; i++) {
    if (flowRapidCount >= FLOW_TIERS[i].at) { tier = i + 1; mult = FLOW_TIERS[i].mult; }
  }
  return { tier, mult };
}
function registerFlowMatch() {
  const now = Date.now();
  const inWindow = (now - lastMatchTime) <= FLOW_WINDOW_MS;
  lastMatchTime = now;
  if (inWindow) {
    flowRapidCount++;
    flowEnergy = Math.min(FLOW_MAX, flowEnergy + FLOW_GAIN_PER_MATCH);
  } else {
    flowRapidCount = 1;
    flowEnergy = Math.min(FLOW_MAX, flowEnergy + FLOW_GAIN_PER_MATCH);
  }
  const prevTier = flowTier;
  const { tier, mult } = recomputeFlowTier();
  flowTier = tier;
  flowMultiplier = mult;
  if (tier > peakFlowTier) peakFlowTier = tier;
  if (tier > prevTier) {
    Sound.playFlowUp(tier);
    showFlowRankup(FLOW_TIERS[tier - 1].name, mult);
  }
  updateFlowUI();
}
function startFlowDecay() {
  stopFlowDecay();
  flowDecayInterval = setInterval(() => {
    if (!gameActive || processing || isReplaying) return;
    if (flowEnergy <= 0 && flowRapidCount === 0) return;
    const idle = Date.now() - lastMatchTime;
    // FIX: honor the 5-second grace window before any decay.
    if (idle > FLOW_GRACE_MS) {
      flowEnergy = Math.max(0, flowEnergy - FLOW_DECAY_PER_SEC * 0.1);
      const impliedCount = Math.round((flowEnergy / FLOW_MAX) * FLOW_MAX);
      if (impliedCount < flowRapidCount) flowRapidCount = impliedCount;
      const prevTier = flowTier;
      const { tier, mult } = recomputeFlowTier();
      flowTier = tier; flowMultiplier = mult;
      if (tier < prevTier) { Sound.playFlowDown(); }
      if (flowEnergy <= 0) { flowRapidCount = 0; flowTier = 0; flowMultiplier = 1; }
      updateFlowUI();
    }
  }, 100);
}
function stopFlowDecay() { clearInterval(flowDecayInterval); flowDecayInterval = null; }
function resetFlow() {
  flowEnergy = 0; flowTier = 0; flowMultiplier = 1;
  flowRapidCount = 0; lastMatchTime = 0; peakFlowTier = 0;
  updateFlowUI();
}

function updateCapsules(matchGroups) {

    capsules.forEach(cap => {
      if (cap.type === group.type && !cap.complete) {
        cap.fill = Math.min(100, cap.fill + group.cells.length * 16);
        if (cap.fill >= 100 && !cap.complete) {
          cap.complete = true;
          capsulesFilled++;

    capsulesFilledThisRun++;
    if (!isReplaying) guildOnCapsuleSetFilled(); // Section 6.1 real contribution

/* Touch/Mouse Handling */
function setupInput() {
  boardEl.addEventListener('touchstart', onTouchStart, { passive: false });
  boardEl.addEventListener('touchmove', onTouchMove, { passive: false });
  boardEl.addEventListener('touchend', onTouchEnd, { passive: false });
  boardEl.addEventListener('touchcancel', onTouchCancel, { passive: false });

  document.addEventListener('mouseup', onMouseUp);

function getCandyAt(x, y) {
  const els = document.elementsFromPoint(x, y);
  for (const el of els) if (el.classList && el.classList.contains('candy')) return el;
  return null;
}
function getCandyRC(el) { return { r: parseInt(el.dataset.row), c: parseInt(el.dataset.col) }; }
function isAdjacent(r1, c1, r2, c2) {
  const dr = Math.abs(r1 - r2), dc = Math.abs(c1 - c2);
  return (dr === 1 && dc === 0) || (dr === 0 && dc === 1);
}
function isValidPos(r, c) { return r >= 0 && r < ROWS && c >= 0 && c < COLS; }
function selectCandy(r, c) {
  clearSelection();
  selectedCandy = { r, c };
  if (candyEls[r] && candyEls[r][c]) candyEls[r][c].classList.add('selected');
}
function clearSelection() {
  if (selectedCandy) {
    const { r, c } = selectedCandy;
    if (candyEls[r] && candyEls[r][c]) candyEls[r][c].classList.remove('selected');
    selectedCandy = null;
  }
}
function clearHints() {
  document.querySelectorAll('.candy.hint').forEach(el => el.classList.remove('hint'));
  clearTimeout(hintTimeout);
}
function startHintTimer() {
  clearTimeout(hintTimeout);
  if (!gameActive || processing || isReplaying) return;
  hintTimeout = setTimeout(() => {
    if (!gameActive || processing) return;
    const move = findValidMove(board);
    if (!move) return;
    const { r1, c1, r2, c2 } = move;
    if (candyEls[r1]?.[c1]) candyEls[r1][c1].classList.add('hint');
    if (candyEls[r2]?.[c2]) candyEls[r2][c2].classList.add('hint');
  }, 5000);
}

function onTouchStart(e) {
  e.preventDefault();
  if (processing || !gameActive || isReplaying) return;
  const touch = e.touches[0];
  const candy = getCandyAt(touch.clientX, touch.clientY);
  if (!candy) return;
  Sound.resume(); Sound.playSelect();
  const { r, c } = getCandyRC(candy);
  touchStartData = { x: touch.clientX, y: touch.clientY, r, c };
  candy.classList.add('touching');
  clearHints();
}
function onTouchMove(e) {
  e.preventDefault();
  if (!touchStartData || processing || !gameActive) return;
  const touch = e.touches[0];
  const dx = touch.clientX - touchStartData.x;
  const dy = touch.clientY - touchStartData.y;
  const dist = Math.sqrt(dx * dx + dy * dy);
  if (dist > SWIPE_THRESHOLD) {
    let dr = 0, dc = 0;
    if (Math.abs(dx) > Math.abs(dy)) dc = dx > 0 ? 1 : -1;
    else dr = dy > 0 ? 1 : -1;
    const { r, c } = touchStartData;
    const r2 = r + dr, c2 = c + dc;
    if (candyEls[r]?.[c]) candyEls[r][c].classList.remove('touching');
    clearSelection();
    const saved = touchStartData;
    touchStartData = null;
    if (isValidPos(r2, c2)) processMove(saved.r, saved.c, r2, c2);
    else Sound.playInvalid();
  }
}
function onTouchEnd(e) {
  e.preventDefault();
  if (!touchStartData) return;
  if (processing || !gameActive) { cleanupTouch(); return; }
  const { r, c } = touchStartData;
  if (candyEls[r]?.[c]) candyEls[r][c].classList.remove('touching');    
if (selectedCandy) {
    const sr = selectedCandy.r, sc = selectedCandy.c;
    if (sr === r && sc === c) clearSelection();
    else if (isAdjacent(sr, sc, r, c)) { clearSelection(); processMove(sr, sc, r, c); }
    else selectCandy(r, c);
  } else selectCandy(r, c);
  touchStartData = null;
}
function onTouchCancel(e) { cleanupTouch(); }
function cleanupTouch() {
  if (touchStartData) {
    const { r, c } = touchStartData;
    if (candyEls[r]?.[c]) candyEls[r][c].classList.remove('touching');
    touchStartData = null;
  }
}
let mouseDown = false, mouseStartData = null;
function onMouseDown(e) {
  if (processing || !gameActive || isReplaying) return;
  const candy = getCandyAt(e.clientX, e.clientY);
  if (!candy) return;
  Sound.resume(); Sound.playSelect();
  mouseDown = true;
  const { r, c } = getCandyRC(candy);
  mouseStartData = { x: e.clientX, y: e.clientY, r, c };
  candy.classList.add('touching');
  clearHints();
}
function onMouseMove(e) {
  if (!mouseDown || !mouseStartData || processing || !gameActive) return;
  const dx = e.clientX - mouseStartData.x;
  const dy = e.clientY - mouseStartData.y;
  const dist = Math.sqrt(dx * dx + dy * dy);
  if (dist > SWIPE_THRESHOLD) {
    let dr = 0, dc = 0;
    if (Math.abs(dx) > Math.abs(dy)) dc = dx > 0 ? 1 : -1;
    else dr = dy > 0 ? 1 : -1;
    const { r, c } = mouseStartData;
    if (candyEls[r]?.[c]) candyEls[r][c].classList.remove('touching');
    clearSelection();
    const saved = mouseStartData;
    mouseStartData = null; mouseDown = false;
    const r2 = saved.r + dr, c2 = saved.c + dc;
    if (isValidPos(r2, c2)) processMove(saved.r, saved.c, r2, c2);
    else Sound.playInvalid();
  }
}
function onMouseUp(e) {
  if (!mouseDown || !mouseStartData) return;
  mouseDown = false;
  if (processing || !gameActive) {
    if (mouseStartData) {
      const { r, c } = mouseStartData;
      if (candyEls[r]?.[c]) candyEls[r][c].classList.remove('touching');
    }
    mouseStartData = null;
    return;
  }
  const { r, c } = mouseStartData;
  if (candyEls[r]?.[c]) candyEls[r][c].classList.remove('touching');
  if (selectedCandy) {
    const sr = selectedCandy.r, sc = selectedCandy.c;
    if (sr === r && sc === c) clearSelection();
    else if (isAdjacent(sr, sc, r, c)) { clearSelection(); processMove(sr, sc, r, c); }
    else selectCandy(r, c);
  } else selectCandy(r, c);
  mouseStartData = null;
}

/* ============================================
   Game Flow
   --------------------------------------------
   processMove records committed swaps into replayLog for live runs and
   skips recording during playback (record=false).
   ============================================ */
async function processMove(r1, c1, r2, c2, record = true) {
  if (processing || !gameActive) return;
  processing = true;
  clearSelection(); clearHints();

  /* 2.3 Record this committed swap (before board mutation) for replays. */
  if (record && !isReplaying && replayLog.length < REPLAY_MAX_SWAPS) {
    replayLog.push({ r1, c1, r2, c2, t: Date.now() - runStartTime });
  }

  /* FIX (replay visibility): highlight the swapped candies during playback
     so a viewer can actually see what the recorded run did. */
  if (isReplaying) {
    if (candyEls[r1]?.[c1]) candyEls[r1][c1].classList.add('replay-swap');
    if (candyEls[r2]?.[c2]) candyEls[r2][c2].classList.add('replay-swap');
    setTimeout(() => {
      document.querySelectorAll('.candy.replay-swap').forEach(el => el.classList.remove('replay-swap'));
    }, 500);
  }

  [board[r1][c1], board[r2][c2]] = [board[r2][c2], board[r1][c1]];
  const el1 = candyEls[r1][c1], el2 = candyEls[r2][c2];
  candyEls[r1][c1] = el2; candyEls[r2][c2] = el1;
  Sound.playSwap();
  updateCandyPos(el1, r2, c2); updateCandyPos(el2, r1, c1);
  await delay(SWAP_MS);

  let isBombSwap = false;
  let matched = new Set(); let groups = [];
  const type1 = board[r1][c1], type2 = board[r2][c2];
  if (type1 === 6 || type2 === 6) {
    isBombSwap = true;
    const bombR = type1 === 6 ? r1 : r2;
    const bombC = type1 === 6 ? c1 : c2;
    const targetR = type1 === 6 ? r2 : r1;
    const targetC = type1 === 6 ? c2 : c1;
    let targetType = board[targetR][targetC];
    if (targetType === -1) targetType = randomType();
    if (targetType === 6) targetType = randomType();
    matched.add(`${bombR},${bombC}`);
    let cells = [{r: bombR, c: bombC}];
    for (let r=0; r<ROWS; r++) for (let c=0; c<COLS; c++) {
      if (board[r][c] === targetType) { matched.add(`${r},${c}`); cells.push({r,c}); }
    }
    groups.push({ type: targetType, cells });
  } else {
    const res = findMatches(board);
    matched = res.matched; groups = res.groups;
  }

  if (matched.size === 0) {
    [board[r1][c1], board[r2][c2]] = [board[r2][c2], board[r1][c1]];
    candyEls[r1][c1] = el1; candyEls[r2][c2] = el2;
    updateCandyPos(el1, r1, c1); updateCandyPos(el2, r2, c2);
    Sound.playInvalid();
    el1.classList.add('invalid-shake'); el2.classList.add('invalid-shake');
    await delay(SWAP_MS);
    el1.classList.remove('invalid-shake'); el2.classList.remove('invalid-shake');
    if (record && !isReplaying && replayLog.length && replayLog[replayLog.length-1].r1 === r1) {
      replayLog.pop();
    }
    processing = false;
    startHintTimer();
    return;
  }

  const synergy = getActiveSynergy();
  const eff = computeLoadoutEffects();
  let cascadeCount = 0;

  lastAchievedMatch = 0;
  groups.forEach(g => { if (g.cells.length > lastAchievedMatch) lastAchievedMatch = g.cells.length; });

  registerFlowMatch();

  while (matched.size > 0) {
    cascadeCount++;
    combo = cascadeCount;
    totalMatches += matched.size;
    if (combo > maxCombo) maxCombo = combo;

    let spawnedBombCoords = null;
    if (!isBombSwap) {
      groups.forEach(g => {
        if (g.cells.length >= 4) {
          let chance = currentPetId === 'galaxy_cat' ? 1.0 : 0.9;
          chance += eff.bombChance;
          if (synergy?.bombBoost) chance = 1.0;
          if (Math.random() <= chance && !spawnedBombCoords) {
            let cell = g.cells.find(c => (c.r === r1 && c.c === c1) || (c.r === r2 && c.c === c2));
            if (!cell) cell = g.cells[0];
            spawnedBombCoords = cell;
          }
        }
      });
    }

    let scoreMult = eff.scoreMult;
    let comboMult = combo * eff.comboMult;
    const points = Math.floor(matched.size * 10 * comboMult * scoreMult * flowMultiplier);
    score += points;

    let coinGain = Math.floor(matched.size * combo * eff.coinMult);
    if (synergy?.coinMult) coinGain *= synergy.coinMult;
    coins += coinGain;

    let extraTime = 1;
    if (currentPetId === 'spirit_fox') extraTime = 2;
    if (synergy?.timePlus) extraTime = synergy.timePlus;
    extraTime += eff.timePerMatchAdd;
    extraTime = Math.max(0, Math.round(extraTime));
    if (extraTime > 0) { timeLeft += extraTime; showFloatingTime(extraTime, matched); }

    updateScore(); updateCombo(); updateCoins(); updateTimer();
    coinsEl.parentElement.style.animation = 'comboPop 0.3s ease';
    setTimeout(() => { coinsEl.parentElement.style.animation = '' }, 300);

    let xpGain = Math.floor(matched.size * 5 * eff.xpMult);
    if (synergy?.xpMult) xpGain *= synergy.xpMult;
    currentXP += xpGain;

    if (currentXP >= xpToNextLevel) {
      playerLevel++;
      currentXP -= xpToNextLevel;
      xpToNextLevel = Math.floor(xpToNextLevel * 1.4);
      const levelBadge = document.getElementById('level-badge');
      if(levelBadge) {
          levelBadge.textContent = `Lv ${playerLevel}`;
          levelBadge.style.animation = 'comboPop 0.5s ease';
          setTimeout(() => levelBadge.style.animation = '', 500);
      }
      const levelReward = 50 + (playerLevel * 10);
      coins += levelReward;
      Sound.playLevelUp();
      showLevelUpEffect();
      saveData();
    }
    updateXPDisplay();

    if (cascadeCount === 1) Sound.playMatch(combo);
    else Sound.playCascade(combo);

    createParticles(matched);
    if (cascadeCount <= 2 || matched.size >= 4) showPraise(matched);
    showScorePopup(matched, points);
    if (cascadeCount >= 2) showComboText(cascadeCount);

    matched.forEach(key => {
      const [mr, mc] = key.split(',').map(Number);
      if (spawnedBombCoords && mr === spawnedBombCoords.r && mc === spawnedBombCoords.c) {
        board[mr][mc] = 6;
        if (candyEls[mr]?.[mc]) {
          candyEls[mr][mc].className = 'candy candy-6';
          candyEls[mr][mc].dataset.type = 6;
        }
      } else if (candyEls[mr]?.[mc]) candyEls[mr][mc].classList.add('matched');
    });

    await delay(MATCH_MS);

    matched.forEach(key => {
      const [mr, mc] = key.split(',').map(Number);
      if (spawnedBombCoords && mr === spawnedBombCoords.r && mc === spawnedBombCoords.c) {}
      else {
        if (candyEls[mr]?.[mc]) { candyEls[mr][mc].remove(); candyEls[mr][mc] = null; }
        board[mr][mc] = -1;
      }
    });

    updateCapsules(groups);

    applyGravity();
    await delay(GRAVITY_MS);
    fillEmpty();
    await delay(SPAWN_MS);

    isBombSwap = false;
    const res = findMatches(board);
    matched = res.matched; groups = res.groups;
    groups.forEach(g => { if (g.cells.length > lastAchievedMatch) lastAchievedMatch = g.cells.length; });
  }

  combo = 1;
  updateCombo();
  if (currentMode === 'freeplay' && !isReplaying && score > bestScore) { bestScore = score; updateBest(); }

  const potential = largestPotentialMatch(board);
  if (potential >= 4 && potential > lastAchievedMatch && potential > bestNearMiss) {
    bestNearMiss = potential;
  }

  if (!hasValidMoves(board)) await shuffleBoard();
  if (!isReplaying) { checkAchievements(); saveData(); }
  processing = false;
  startHintTimer();
}

async function shuffleBoard() {
  for (let r = 0; r < ROWS; r++) for (let c = 0; c < COLS; c++) {
    if (candyEls[r]?.[c]) candyEls[r][c].classList.add('shuffle-out');
  }
  await delay(350);
  boardEl.querySelectorAll('.candy').forEach(el => el.remove());
  board = generateBoard();
  candyEls = [];
  for (let r = 0; r < ROWS; r++) {
    candyEls[r] = [];
    for (let c = 0; c < COLS; c++) {
      const el = createCandyEl(board[r][c], -(ROWS - r), c);
      candyEls[r][c] = el;
    }
  }
  boardEl.offsetHeight;
  for (let r = 0; r < ROWS; r++) for (let c = 0; c < COLS; c++) {
    setTimeout(() => updateCandyPos(candyEls[r][c], r, c), 20 + r * 40 + c * 15);
  }
  await delay(ROWS * 60 + 300);
}

/* ============================================
   2.3 REPLAY PLAYBACK (FIXED)
   --------------------------------------------
   FIX 1: A "Quit Replay" button + progress readout are injected so the
          player can abort playback at any time.
   FIX 2: The board is rebuilt from the EXACT recorded initial state
          (seeded regen for challenge, saved initialBoard for freeplay) and
          each swap is replayed through the real processMove animation path,
          so the replay actually shows what happened in the game.
   ============================================ */
function injectReplayControls() {
  removeReplayControls();
  const quit = document.createElement('button');

async function playReplay(replay, mode) {
  if (!replay || !replay.log || !replay.log.length) {
    showRewardToast('No replay recorded yet');
    return;
  }
  hideAllModals();
  document.getElementById('start-screen').classList.add('hidden');

  isReplaying = true;
  replayCancelled = false;
  gameActive = true;
  processing = false;
  currentMode = mode;
  score = 0; combo = 1; timeLeft = GAME_TIME;
  resetFlow();
  updateScore(); updateCombo(); updateTimer(); updateCoins();
  calculateLayout();

  document.getElementById('board-inner').classList.add('replay-mode');
  if (mode === 'challenge') document.getElementById('board-frame').classList.add('challenge-mode');
  else document.getElementById('board-frame').classList.remove('challenge-mode');

  injectReplayControls();

  /* Rebuild the EXACT starting board (fix for "display what happened"). */
  if (mode === 'challenge' && replay.seed != null) {
    seededRNG = mulberry32(replay.seed);
    board = generateBoard();
    seededRNG = mulberry32(replay.seed); // re-arm so initial capsules match too
    renderFullBoard();
    initCapsules();
    seededRNG = null; // cascades used Math.random live, so match that here
  } else if (replay.initialBoard) {
    board = replay.initialBoard.map(row => row.slice());
    renderFullBoard();
    initCapsules();
  } else {
    board = generateBoard();
    renderFullBoard();
    initCapsules();
  }

  const startedAt = Date.now();
  for (let i = 0; i < replay.log.length; i++) {
    if (replayCancelled || !isReplaying) break;
    const swap = replay.log[i];
    const targetElapsed = swap.t / REPLAY_SPEED;
    const waitMs = targetElapsed - (Date.now() - startedAt);
    if (waitMs > 0) await delay(waitMs);
    if (replayCancelled || !isReplaying) break;
    updateReplayProgress(i + 1, replay.log.length);
    await processMoveReplay(swap.r1, swap.c1, swap.r2, swap.c2);
  }

  if (replayCancelled) return;
  await delay(600);
  finishReplay();
}

/* Playback wrapper that ignores the live-input guards. */
async function processMoveReplay(r1, c1, r2, c2) {
  const wasActive = gameActive;
  gameActive = true;
  processing = false;
  await processMove(r1, c1, r2, c2, false);
  gameActive = wasActive;
}

function finishReplay() {

  gameActive = false;
  processing = false;

  document.getElementById('board-inner').classList.remove('replay-mode');
  document.getElementById('board-frame').classList.remove('challenge-mode');
  showRewardToast(`▶ Replay finished · ${score} pts`);
  setTimeout(() => showStartScreen(), 900);
}

/* Save the just-finished run as a personal-best replay if it beat the prior
   best for its mode (Section 2.3). */
function maybeSaveReplay(finalScore, mode) {
  const duration = Math.round((Date.now() - runStartTime) / 1000);
  if (mode === 'challenge') {
    const today = getUTCDateKey();
    if (!bestChallengeReplay || bestChallengeReplay.date !== today || finalScore > bestChallengeReplay.score) {
      bestChallengeReplay = {
        date: today, score: finalScore, seed: currentRunSeed,
        log: replayLog.slice(), duration,
      };
    }
  } else {
    if (!bestFreeplayReplay || finalScore > bestFreeplayReplay.score) {
      bestFreeplayReplay = {
        score: finalScore, log: replayLog.slice(),
        initialBoard: currentInitialBoard ? currentInitialBoard.map(row => row.slice()) : null,
        duration,
      };
    }
  }
  saveData();
}

/* Timer */
function startTimer() {
  stopTimer();
  timerInterval = setInterval(() => {
    if (!gameActive) return;
    if (isReplaying) return;
    if (infinityTimeUntil > Date.now()) { updateTimer(); return; }
    const eff = computeLoadoutEffects();
    if (eff.timerSlow) {
      if (!startTimer._tick) startTimer._tick = false;
      startTimer._tick = !startTimer._tick;
      if (startTimer._tick) timeLeft--;
    } else timeLeft--;
    updateTimer();
    if (timeLeft <= 10 && timeLeft > 0) Sound.playTick();
    if (timeLeft <= 0) {
      stopTimer();
      gameActive = false;
      processing = true;
      Sound.playTimeUp();
      endRun();
    }
  }, 1000);
}
function stopTimer() { clearInterval(timerInterval); timerInterval = null; }

/* ============================================
   END-OF-RUN handling (Section 2)
   ============================================ */
function endRun() {
  stopFlowDecay();
  stopGhostOverlay();
  const finalScore = score;
  const mode = currentMode;

  totalGames++;
  submitScore(finalScore, mode);

  if (mode === 'challenge') {
    const today = getUTCDateKey();
    if (challengeDate !== today) { challengeDate = today; challengeBestScore = 0; challengeAttempts = 0; }
    challengeAttempts++;
    if (finalScore > challengeBestScore) challengeBestScore = finalScore;
  }

  maybeSaveReplay(finalScore, mode);

  saveData();
  checkAchievements();
  updateBest();
  setTimeout(() => showTimeUpModal(), 500);
}

/* ============================================
   WHEEL OF FORTUNE
   ============================================ */
function buildWheelSVG() {
  const svg = document.getElementById('wheel-svg');
  if (!svg) return;
  svg.innerHTML = '';
  const cx = 100, cy = 100, r = 95;
  const n = WHEEL_PRIZES.length;
  const step = (Math.PI * 2) / n;

  const outer = document.createElementNS('http://www.w3.org/2000/svg','circle');
  outer.setAttribute('cx', cx); outer.setAttribute('cy', cy);
  outer.setAttribute('r', r + 3);
  outer.setAttribute('fill', '#fff');
  outer.setAttribute('stroke', '#a78bfa');
  outer.setAttribute('stroke-width', '3');
  svg.appendChild(outer);

  for (let i = 0; i < n; i++) {
    const startA = -Math.PI/2 + i * step;
    const endA = startA + step;
    const x1 = cx + r * Math.cos(startA);
    const y1 = cy + r * Math.sin(startA);
    const x2 = cx + r * Math.cos(endA);
    const y2 = cy + r * Math.sin(endA);
    const path = document.createElementNS('http://www.w3.org/2000/svg','path');
    const largeArc = step > Math.PI ? 1 : 0;
    const d = `M ${cx} ${cy} L ${x1} ${y1} A ${r} ${r} 0 ${largeArc} 1 ${x2} ${y2} Z`;
    path.setAttribute('d', d);
    path.setAttribute('fill', WHEEL_PRIZES[i].color);
    path.setAttribute('stroke', '#fff');
    path.setAttribute('stroke-width', '2');
    svg.appendChild(path);

    const midA = startA + step / 2;
    const tr = r * 0.62;
    const tx = cx + tr * Math.cos(midA);
    const ty = cy + tr * Math.sin(midA);
    const text = document.createElementNS('http://www.w3.org/2000/svg','text');
    text.setAttribute('x', tx);
    text.setAttribute('y', ty);
    text.setAttribute('text-anchor', 'middle');
    text.setAttribute('dominant-baseline', 'middle');
    text.setAttribute('font-size', WHEEL_PRIZES[i].label.length > 6 ? '9' : '12');
    const degrees = (midA * 180 / Math.PI) + 90;
    text.setAttribute('transform', `rotate(${degrees}, ${tx}, ${ty})`);
    const parts = WHEEL_PRIZES[i].label.split(' ');
    if (parts.length > 1) {
      const t1 = document.createElementNS('http://www.w3.org/2000/svg','tspan');
      t1.setAttribute('x', tx); t1.setAttribute('dy', '-0.5em'); t1.textContent = parts[0];
      const t2 = document.createElementNS('http://www.w3.org/2000/svg','tspan');
      t2.setAttribute('x', tx); t2.setAttribute('dy', '1.1em'); t2.textContent = parts.slice(1).join(' ');
      text.appendChild(t1); text.appendChild(t2);

function canSpinWheel() { return (Date.now() - lastWheelSpin) >= WHEEL_COOLDOWN_MS; }

function formatCountdown(ms) {
  if (ms <= 0) return 'Ready!';
  const totalSec = Math.floor(ms / 1000);
  const h = Math.floor(totalSec / 3600);
  const m = Math.floor((totalSec % 3600) / 60);
  const s = totalSec % 60;
  return `${String(h).padStart(2,'0')}:${String(m).padStart(2,'0')}:${String(s).padStart(2,'0')}`;
}

function spinWheel() {
  if (wheelSpinning) return;
  if (!canSpinWheel()) { Sound.playInvalid(); return; }
  wheelSpinning = true;
  const spinBtn = document.getElementById('wheel-spin-btn');
  spinBtn.disabled = true;
  spinBtn.textContent = '...';
  Sound.playSpin();

  const n = WHEEL_PRIZES.length;
  const step = 360 / n;
  const prizeIdx = rand(0, n - 1);
  const targetAngle = -(prizeIdx * step + step / 2);
  const currentMod = ((wheelCurrentRotation % 360) + 360) % 360;
  const targetMod = ((targetAngle % 360) + 360) % 360;
  let deltaForward = (targetMod - currentMod + 360) % 360;
  const extraSpins = 5;
  const totalDelta = deltaForward + 360 * extraSpins;
  wheelCurrentRotation += totalDelta;

  const svg = document.getElementById('wheel-svg');
  svg.style.transform = `rotate(${wheelCurrentRotation}deg)`;

  setTimeout(() => {
    const prize = WHEEL_PRIZES[prizeIdx];
    applyWheelPrize(prize);
    lastWheelSpin = Date.now();
    saveData();
    Sound.playWin();
    wheelSpinning = false;
    spinBtn.disabled = true;
    spinBtn.classList.add('disabled');
    spinBtn.textContent = 'WAIT';
    updateEventUI();
  }, 4100);
}

function applyWheelPrize(prize) {
  if (prize.type === 'coins') {
    coins += prize.amount; updateCoins();
    showRewardToast(`🪙 +${prize.amount}`);
  } else if (prize.type === 'infinity') {
    infinityTimeUntil = Date.now() + prize.amount * 1000;
    updateTimer();
    showRewardToast(`⏳ INFINITY TIME 1H!`);
  }
}

}

function setupEventCarousel() {
  const carousel = document.getElementById('events-carousel');
  const dots = document.querySelectorAll('.event-dot');
  if (!carousel) return;
  carousel.addEventListener('scroll', () => {
    const idx = Math.round(carousel.scrollLeft / carousel.clientWidth);
    dots.forEach((d, i) => d.classList.toggle('active', i === idx));
  });
  dots.forEach(dot => {
    dot.addEventListener('click', () => {
      const idx = parseInt(dot.dataset.idx);
      carousel.scrollTo({ left: idx * carousel.clientWidth, behavior: 'smooth' });
    });
  });
}

function startGame(mode = 'freeplay') {
  Sound.init(); Sound.resume(); Sound.playStart();
  hideStartScreen();
  hideChallengeModal();
  currentMode = mode;
  replayCancelled = false;

  if (mode === 'challenge') {
    const key = getUTCDateKey();
    currentRunSeed = hashStringToSeed('EmberChain-' + key);
    seededRNG = mulberry32(currentRunSeed);
    document.getElementById('board-frame').classList.add('challenge-mode');
  } else {
    currentRunSeed = null;
    seededRNG = null;
    document.getElementById('board-frame').classList.remove('challenge-mode');
  }

  score = 0;
  let initialTime = GAME_TIME;
  if (mode === 'freeplay') {
    if (currentPetId === 'unicorn') initialTime += 10;
    if (currentPetId === 'wizard_dog') initialTime += 30;
  }
  timeLeft = initialTime;
  combo = 1; processing = false; gameActive = true; isReplaying = false;
  resetFlow();
  bestNearMiss = 0;
  lastAchievedMatch = 0;
  capsulesFilledThisRun = 0;

  replayLog = [];
  runStartTime = Date.now();

  updateScore(); updateTimer(); updateCombo(); updateCoins(); updateXPDisplay(); updateBest();
  calculateLayout();
  board = generateBoard();

  /* Snapshot the exact starting board so freeplay replays are faithful. */
  currentInitialBoard = board.map(row => row.slice());

  /* For challenge, re-arm the seed so capsule colors also match, then disarm
     so cascades/refills use Math.random (mirrored in replay). */
  if (mode === 'challenge') {
    seededRNG = mulberry32(currentRunSeed);
  }

  renderFullBoard();
  initCapsules();
  seededRNG = null;

  const today = getUTCDateKey();
  if (mode === 'challenge' && ghostEnabled && bestChallengeReplay &&
      bestChallengeReplay.date === today && bestChallengeReplay.log.length) {
    ghostLog = bestChallengeReplay.log;
    startGhostOverlay();
  } else {
    ghostLog = null;
  }

  startTimer();
  startFlowDecay();
  startHintTimer();
  createConfetti();
}
